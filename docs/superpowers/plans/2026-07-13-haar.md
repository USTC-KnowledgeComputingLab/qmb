"""Loss functions for local optimization in HAAR algorithm."""

from __future__ import annotations

import jax.numpy as jnp
from jax import Array


def sum_filtered_angle_scaled_log(psi_net: Array, psi_target: Array) -> Array:
    """Sum of filtered angle-scaled log differences.

    Measures the angle between network and target wavefunctions, weighted
    by target amplitude to focus on dominant configurations.
    """
    target_prob = (psi_target.conj() * psi_target).real
    target_weight = target_prob / (jnp.sum(target_prob) + 1e-30)

    angle_diff = (psi_net.conj() * psi_target).real / (
        jnp.abs(psi_net) * jnp.abs(psi_target) + 1e-30
    )
    angle_diff = jnp.clip(angle_diff, -1.0, 1.0)
    angle = jnp.arccos(angle_diff)

    return jnp.sum(target_weight * angle**2)
```

import jax
import jax.numpy as jnp
from jax import Array

def _lanczos_init(psi: Array) -> tuple[Array, list[Array], list[Array], list[Array]]:
    v0 = psi / jnp.linalg.norm(psi)
    return v0, [], [], [v0]


def _lanczos_step(
    model: object,
    configs: Array,
    v: list[Array],
    alpha: list[Array],
    w: Array | None,
    prev_beta: float = 0.0,
) -> tuple[list[Array], list[Array], Array, float]:
    if w is None:
        w = _apply_hamiltonian(model, configs, v[-1], configs)
        a = (w.conj() @ v[-1]).real
        alpha.append(a)
        w = w - a * v[-1]
        return v, alpha, w, 0.0

    norm_w = jnp.linalg.norm(w)
    v_next = w / norm_w
    v.append(v_next)
    beta = norm_w

    w = _apply_hamiltonian(model, configs, v[-1], configs)
    a = (w.conj() @ v[-1]).real
    alpha.append(a)
    w = w - a * v[-1] - prev_beta * v[-2]

    for prev_v in v:
        w = w - (prev_v.conj() @ w) * prev_v

    return v, alpha, w, beta


def _apply_hamiltonian(model: object, configs_i: Array, psi_i: Array, configs_j: Array) -> Array:
    psi_i_real = jnp.stack([psi_i.real, psi_i.imag], axis=1)
    h_psi = model.apply_within_subspace(configs_i, psi_i_real, configs_j)
    return h_psi[:, 0] + 1j * h_psi[:, 1]
```

```python
def _eigh_tridiagonal(
    alpha: list[Array], beta: list[Array], v: list[Array], state_count: int
) -> list[tuple[float, Array]]:
    if len(beta) == 0:
        return [(float(alpha[0]), v[0])]
    n = len(alpha)
    h_tri = jnp.zeros((n, n), dtype=jnp.float64)
    h_tri = h_tri.at[jnp.arange(n), jnp.arange(n)].set(jnp.array([float(a) for a in alpha]))
    h_tri = h_tri.at[jnp.arange(n - 1), jnp.arange(1, n)].set(jnp.array([float(b) for b in beta]))
    h_tri = h_tri.at[jnp.arange(1, n), jnp.arange(n - 1)].set(jnp.array([float(b) for b in beta]))
    vals, vecs = jnp.linalg.eigh(h_tri)
    k = min(state_count, n)
    results = []
    for i in range(k):
        psi = sum(float(vecs[j, i]) * v[j] for j in range(n))
        results.append((float(vals[i]), psi))
    return results
```

### Task 4: Sampling helpers

```python
def _sample_from_pool(pool: tuple[Array, Array, Array] | None, count: int, key: Array) -> tuple[Array, Array]:
    if pool is None:
        return jnp.zeros((0, 1), dtype=jnp.uint8), jnp.zeros((0,), dtype=jnp.complex128)
    configs, psi, _counts = pool
    prob = (psi.conj() * psi).real
    prob = prob / (jnp.sum(prob) + 1e-30)
    log_prob = jnp.log(prob + 1e-30)
    g = log_prob + jax.random.gumbel(key, log_prob.shape)
    _, indices = jax.lax.top_k(g, min(count, len(g)))
    return configs[indices], psi[indices]


def _merge_pools(
    configs_a: Array, psi_a: Array, configs_b: Array, psi_b: Array
) -> tuple[Array, Array]:
    if configs_b.shape[0] == 0:
        return configs_a, psi_a
    both_c = jnp.concatenate([configs_a, configs_b], axis=0)
    both_p = jnp.concatenate([psi_a, psi_b], axis=0)
    _, idx, counts = jnp.unique(
        both_c.view(jnp.uint32).reshape(both_c.shape[0], -1), axis=0,
        return_index=True, return_counts=True, size=both_c.shape[0], fill_value=0
    )
    # dedup: prefer psi_b (from pool) when both have same config
    result_c = both_c[idx]
    result_p = both_p[idx]
    return result_c, result_p
```

    import pickle

        with open(path, "rb") as f:

    except FileNotFoundError:
